JS — Data-driven re-render pattern

JS 
// 1. State array — mỗi object đại diện 1 dòng
var conditionRows = [];

// 2. Build HTML string cho từng cell (dùng sd-components)
function buildParamHtml(idx, selectedVal) {
  var items = JSON.stringify([
    { id: 'loai_hinh', name: 'Loại hình' },
    { id: 'trang_thai', name: 'Trạng thái' }
    // ...
  ]);
  return '<sd-select size="sm" placeholder="Chọn thông số" ' +
    'items=\'' + items + '\' ' +
    (selectedVal ? 'value="' + selectedVal + '"' : '') +
    ' data-cond-param="' + idx + '"></sd-select>';
}

function buildDelHtml(idx) {
  return '<sd-button type="link" color="secondary" size="sm" ' +
    'prefixIcon="close" aria-label="Xóa" ' +
    'data-cond-del="' + idx + '"></sd-button>';
}

// 3. Re-render toàn bộ table bằng cách set attribute data
function renderConditionTable() {
  var tableData = conditionRows.map(function(r, i) {
    return {
      _param: buildParamHtml(i, r.param),
      _op:    buildOpHtml(i, r.type, r.op),
      _val:   buildValHtml(i, r.param, r.op),
      _del:   buildDelHtml(i)
    };
  });
  document.getElementById('condition-table')
    .setAttribute('data', JSON.stringify(tableData));
}

// 4. Wire events
document.addEventListener('DOMContentLoaded', function () {
  var table = document.getElementById('condition-table');

  // Thêm dòng mới
  table.addEventListener('sd-table-add-row', function () {
    conditionRows.push({ param: '', op: '', type: 'text' });
    renderConditionTable();
  });

  // Lắng nghe thay đổi trong cell (sd-select, sd-input đều fire sd-change)
  table.addEventListener('sd-change', function (e) {
    var paramIdx = e.target.getAttribute('data-cond-param');
    if (paramIdx !== null) {
      var i = parseInt(paramIdx, 10);
      conditionRows[i].param = e.target.value;
      conditionRows[i].op    = '';          // reset op khi đổi param
      renderConditionTable();
      return;
    }
    var opIdx = e.target.getAttribute('data-cond-op');
    if (opIdx !== null) {
      conditionRows[parseInt(opIdx, 10)].op = e.target.value;
      renderConditionTable();
    }
  });

  // Xóa dòng
  table.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-cond-del]');
    if (!btn) return;
    conditionRows.splice(parseInt(btn.getAttribute('data-cond-del'), 10), 1);
    renderConditionTable();
  });
});

Lưu ý kỹ thuật

  • overflow: visible — PHẢI override .c-container và .c-table trên element cụ thể (scoped selector). Không override global để tránh ảnh hưởng sd-table khác trên trang.
  • Re-render toàn bộ — mỗi lần state thay đổi, set lại data attribute. sd-table sẽ re-render innerHTML. Đây là cách duy nhất vì sd-table là display-only web component.
  • Event delegation — lắng nghe trên sd-table element, không phải từng cell. Dùng data-* attribute trên sd-components bên trong cell để identify row/column.
  • sd-change event — tất cả sd-form components (sd-select, sd-input) đều fire sd-change (bubbles: true) khi giá trị thay đổi.
  • title: "" — dùng title: "" (empty string) cho cột delete để ẩn header. sd-table xử lý đúng (title != null ? title : field).

HTML 
<sd-table hide-inline-filter paginated pagesize="10"
  option='{"filter":{"quickSearch":{
      "containFields":["code","name"],
      "equalFields":["taxCode"],
      "placeholder":"Tìm theo mã hoặc tên…",
      "filters":[
        {"field":"bank","title":"Ngân hàng","type":"values","required":true,
         "option":{"items":[…],"valueField":"id","displayField":"name"}},
        {"field":"status","title":"Trạng thái","type":"values","default":["pending"],
         "option":{"items":[…],"valueField":"id","displayField":"name","selection":"MULTIPLE"}}
      ]}},
    "aggregate":{"scope":"filtered"}}'
  columns='[
    {"field":"code","title":"Mã","type":"text","aggregate":"COUNT"},
    {"field":"qty","title":"Số lượng","type":"number","align":"right","aggregate":"SUM"},
    {"field":"date","title":"Ngày","type":"date","aggregate":"MAX"}
  ]'
  data='[…]'></sd-table>

Hành vi

  • containFields: từ khoá CHỨA (không phân biệt hoa thường) ở một trong các field · equalFields: từ khoá BẰNG ĐÚNG — hai danh sách gộp OR; không khai field nào thì từ khoá bị bỏ qua. placeholder mặc định "Tìm kiếm…". aggregate.scope: page (mặc định, các dòng đang hiện) hoặc filtered (mọi dòng sau lọc).
  • Dropdown áp ngay (một giá trị = bằng, chọn nhiều = thuộc tập), AND giữa các dropdown và với lọc cột / Bộ lọc ngoài; từ khoá áp khi Enter, ✕ xoá và áp luôn; mỗi lần đổi về trang 1.
  • required chưa chọn → ô từ khoá khoá + gợi ý, bảng rỗng "Vui lòng chọn bộ lọc để bắt đầu". Chế độ server (items là hàm) nhận filter.quickSearch = { term, filters } và không được gọi khi thiếu dropdown bắt buộc.
  • aggregate: SUM / AVERAGE (cột number), MIN / MAX (number, date, datetime), COUNT (mọi cột), hoặc hàm (items, ctx) => giá trị qua property. Không tính được → ---.
  • Angular: cùng tên khoá trong SdTableOption của @sdcorejs/angular 22.2.14 — dự án còn ở 19.0.10 thì không có (ai-instructions/html-to-angular.md §0).
  • Bảng cây: aggregate.tree = { items: 'leaves' | 'roots' | 'all', subtotal: true }; tìm nhanh giữ nhánh có node khớp, tự mở node có con cháu khớp.
  • Nhóm: group = { fields, collapsible, defaultCollapsed, template(ctx) } + aggregate.group: true. template là của kit (thay <ng-template sdGroupDef>), nhận { key, values, items, isExpanded }, trả HTML; không khai thì nhãn là JSON giá trị nhóm (như thư viện).
  • Export: export = { visible: 'ALL' | 'EXCEL' | 'CSV', fileName, columns, mapping, items } hoặc { type: 'custom', onExport(filter) }; xuất thật file (exceljs / CSV UTF-8). Cột export: { disabled: true } bị bỏ; transform(value, row, { isExport: true }).
  • Nhóm hành động: selector.actions[] = { variant: 'group', title, icon, children: [ { title, icon, click(rows) } ] }.
